####r2 function for evaluating models:
r2.mixed<-function(mF){
  mFX<-model.matrix(mF)
  VarF <- var(as.vector(fixef(mF) %*% t(mFX)))
  VarR<-sum(as.numeric(VarCorr(mF))) 
  VarResid<-attr(VarCorr(mF), "sc")^2
  fR2<-VarF/(VarF + VarR + VarResid)
  rfR2<-(VarF + VarR)/(VarF + VarR + VarResid)
  list(fR2=fR2,rfR2=rfR2)
}


####vif function for evaluating models:

vif.mer <- function (fit) {
  ## adapted from rms::vif
  
  v <- vcov(fit)
  nam <- names(fixef(fit))
  
  ## exclude intercepts
  ns <- sum(1 * (nam == "Intercept" | nam == "(Intercept)"))
  if (ns > 0) {
    v <- v[-(1:ns), -(1:ns), drop = FALSE]
    nam <- nam[-(1:ns)]
  }
  
  d <- diag(v)^0.5
  v <- diag(solve(v/(d %o% d)))
  names(v) <- nam
  v
}

####Multiple linear regression models (MLRMs) used to assess the main and interactive effects of substrate availability, thermal adaptation and MAT on on soil microbial respiration, growth, and carbon use efficiency.

####R packages will be used for modelling analysis:
library(broom)
library(dplyr)
library(tidyr)
library(writexl)
library(sandwich)
library(lmtest)

##Clear the environment:
rm(list = ls())

# Load data:
dat<-read.csv("MLRMs.csv",header=T,na.strings=c("NA"))

# Define variables:
vars <- c("MAT", "Substrate", "Adaptation", "Respiration", "Growth", "CUE")

# Clean and transform data:
dat <- dat %>%
  mutate(across(all_of(vars), as.numeric)) %>%
  filter(if_all(all_of(vars), ~ !is.na(.))) %>%
  mutate(c_MAT = MAT - mean(MAT), Respiration_z = as.numeric(scale(Respiration)), Growth_z = as.numeric(scale(Growth)), CUE_z = as.numeric(scale(CUE)))

# Fit standardized response models:
m_Respiration_z <- lm(Respiration_z ~ Substrate * Adaptation * c_MAT, data = dat)
m_Growth_z <- lm(Growth_z ~ Substrate * Adaptation * c_MAT, data = dat)
m_CUE_z <- lm(CUE_z ~ Substrate * Adaptation * c_MAT, data = dat)

# Extract HC3 robust results:
extract_hc3_results <- function(model, response_name) {
  vcov_hc3 <- vcovHC(model, type = "HC3")
  coef_table <- tidy(coeftest(model, vcov. = vcov_hc3))
  ci_table <- as.data.frame(coefci(model, vcov. = vcov_hc3, level = 0.95))
  ci_table$term <- rownames(ci_table)
  names(ci_table)[1:2] <- c("conf.low", "conf.high")
  coef_table %>%
    left_join(ci_table, by = "term") %>%
    mutate(Response = response_name, R2 = summary(model)$r.squared, Adjusted_R2 = summary(model)$adj.r.squared, Response_scale = "Standardized response: z-score") %>%
    select(Response, Response_scale, Term = term, Estimate = estimate, `Std. Error` = std.error, `t value` = statistic, `Pr(>|t|)` = p.value, R2, Adjusted_R2, conf.low, conf.high)
}

# Combine model results:
coef_all_z <- bind_rows(extract_hc3_results(m_Respiration_z, "Respiration"), extract_hc3_results(m_Growth_z, "Growth"), extract_hc3_results(m_CUE_z, "CUE"))

# Separate result tables:
Respiration_table_z <- filter(coef_all_z, Response == "Respiration")
Growth_table_z <- filter(coef_all_z, Response == "Growth")
CUE_table_z <- filter(coef_all_z, Response == "CUE")
